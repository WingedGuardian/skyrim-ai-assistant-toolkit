cmake_minimum_required(VERSION 3.24)
project(ToolkitProbe VERSION 0.1.0 LANGUAGES CXX)
option(PROBE_PLUGIN "Build Windows SKSE plugin" ON)
enable_testing()
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(probe_logic_tests tests/logic_tests.cpp)
target_include_directories(probe_logic_tests PRIVATE src)
target_compile_features(probe_logic_tests PRIVATE cxx_std_23)
add_test(NAME probe_logic COMMAND probe_logic_tests)
add_executable(probe_contract_tests tests/contract_tests.cpp)
target_include_directories(probe_contract_tests PRIVATE src)
target_link_libraries(probe_contract_tests PRIVATE nlohmann_json::nlohmann_json)
target_compile_features(probe_contract_tests PRIVATE cxx_std_23)
add_test(NAME probe_contract COMMAND probe_contract_tests)
if(PROBE_PLUGIN)
 set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
 add_subdirectory(${COMMONLIB_SOURCE_DIR} commonlib)
 include(${COMMONLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake)
 add_commonlibsse_plugin(ToolkitProbe AUTHOR "Skyrim AI Toolkit contributors" SOURCES src/plugin.cpp vendor/DevBenchAPI.cpp)
 target_include_directories(ToolkitProbe PRIVATE vendor src)
 target_link_libraries(ToolkitProbe PRIVATE nlohmann_json::nlohmann_json)
 target_compile_features(ToolkitProbe PRIVATE cxx_std_23)
 target_compile_options(ToolkitProbe PRIVATE /Zc:preprocessor /permissive- /EHsc)
 target_precompile_headers(ToolkitProbe PRIVATE src/PCH.h)
endif()
